typescriptasync function getTenantAddressContext(tenantDb: TenantDb): Promise<{
  city?: string;
  state?: string;
  zipPrefix?: string;
}> {
  try {
    const rows = await tenantDb.select().from(businessSettings).limit(1);
    const row = rows[0];
    if (!row) return {};

    // Read new dedicated columns first:
    const city = (row as any).businessCity;
    const state = (row as any).businessState;
    const zipPrefix = (row as any).businessZipPrefix;
    if (city && state) return { city: city.trim(), state: state.trim(), zipPrefix: zipPrefix?.trim() };

    return {};
  } catch (err) {
    console.warn('[MAPS] getTenantAddressContext fail-open:', err);
    return {};
  }
}
